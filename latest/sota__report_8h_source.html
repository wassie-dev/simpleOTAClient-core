92static inline uint32_t sota_jitter_s(uint32_t wait_s, uint32_t random)
93{
94 if (wait_s < SOTA_JITTER_MIN_S) return wait_s;
95 const uint32_t span = wait_s / 5u;
96 const uint64_t jittered =
97 (uint64_t)wait_s - span / 2u + (((uint64_t)random * ((uint64_t)span + 1u)) >> 32);
98 return jittered > UINT32_MAX ? UINT32_MAX : (uint32_t)jittered;
99}

101/*
102 * Outbox. Once a boot has worked out what happened to a trial, the report
103 * moves here and the trial keys are cleared, so a later update never
104 * overwrites a report that has not been sent. Stored as one NVS blob. When

106 */

150static inline bool sota_outbox_valid(const sota_outbox_t *box)
151{
152 if (box->version != SOTA_OUTBOX_VERSION || box->count > SOTA_OUTBOX_SLOTS) return false;
153 for (uint8_t i = 0; i < box->count; ++i) {
154 const sota_report_t *r = &box->reports[i];
155 if (!sota_field_terminated(r->event, sizeof(r->event)) ||
156 !sota_field_terminated(r->reason, sizeof(r->reason)) ||
157 !sota_field_terminated(r->deployment, sizeof(r->deployment)) ||
158 !r->event[0] || !r->deployment[0] || r->build == 0) {
159 return false;
160 }
161 }
162 return true;
163}

167static inline bool sota_outbox_push(sota_outbox_t *box, const char *event,
168 const char *reason, const char *deployment,
169 uint32_t build)
170{
171 sota_report_t r;
172 memset(&r, 0, sizeof(r));
173 if (!event || !event[0] || !deployment || !deployment[0] || build == 0) return false;
174 if (!sota_copy_field(r.event, sizeof(r.event), event) ||
175 !sota_copy_field(r.reason, sizeof(r.reason), reason) ||
176 !sota_copy_field(r.deployment, sizeof(r.deployment), deployment)) {
177 return false;
178 }
179 r.build = build;
180 if (box->count == SOTA_OUTBOX_SLOTS) {
181 memmove(&box->reports[0], &box->reports[1],
182 sizeof(sota_report_t) * (SOTA_OUTBOX_SLOTS - 1));
183 box->count--;
184 }
185 box->reports[box->count++] = r;
186 return true;
187}

190static inline void sota_outbox_pop(sota_outbox_t *box)
191{
192 if (box->count == 0) return;
193 memmove(&box->reports[0], &box->reports[1],
194 sizeof(sota_report_t) * (SOTA_OUTBOX_SLOTS - 1));
195 box->count--;
196 memset(&box->reports[box->count], 0, sizeof(sota_report_t));
197}
