57/*
58 * Outbox. Once a boot has worked out what happened to a trial, the report
59 * moves here and the trial keys are cleared, so a later update never
60 * overwrites a report that has not been sent. Stored as one NVS blob. When
61 * full, the oldest report makes room for the new one.
62 */

106static inline bool sota_outbox_valid(const sota_outbox_t *box)
107{
108 if (box->version != SOTA_OUTBOX_VERSION || box->count > SOTA_OUTBOX_SLOTS) return false;
109 for (uint8_t i = 0; i < box->count; ++i) {
110 const sota_report_t *r = &box->reports[i];
111 if (!sota_field_terminated(r->event, sizeof(r->event)) ||
112 !sota_field_terminated(r->reason, sizeof(r->reason)) ||
113 !sota_field_terminated(r->deployment, sizeof(r->deployment)) ||
114 !r->event[0] || !r->deployment[0] || r->build == 0) {
115 return false;
116 }
117 }
118 return true;
119}

123static inline bool sota_outbox_push(sota_outbox_t *box, const char *event,
124 const char *reason, const char *deployment,
125 uint32_t build)
126{
127 sota_report_t r;
128 memset(&r, 0, sizeof(r));
129 if (!event || !event[0] || !deployment || !deployment[0] || build == 0) return false;
130 if (!sota_copy_field(r.event, sizeof(r.event), event) ||
131 !sota_copy_field(r.reason, sizeof(r.reason), reason) ||
132 !sota_copy_field(r.deployment, sizeof(r.deployment), deployment)) {
133 return false;
134 }
135 r.build = build;
136 if (box->count == SOTA_OUTBOX_SLOTS) {
137 memmove(&box->reports[0], &box->reports[1],
138 sizeof(sota_report_t) * (SOTA_OUTBOX_SLOTS - 1));
139 box->count--;
140 }
141 box->reports[box->count++] = r;
142 return true;
143}

146static inline void sota_outbox_pop(sota_outbox_t *box)
147{
148 if (box->count == 0) return;
149 memmove(&box->reports[0], &box->reports[1],
150 sizeof(sota_report_t) * (SOTA_OUTBOX_SLOTS - 1));
151 box->count--;
152 memset(&box->reports[box->count], 0, sizeof(sota_report_t));
153}
