84static inline sota_boot_outcome_t sota_classify_boot(const sota_boot_facts_t *f)
85{
86 /* sota_staged is written first and sota_trial last, so staged with no
87 * trial is a record cut short before the slot switch. */
88 if (f->trial == 0) return f->staged ? SOTA_BOOT_ABANDONED : SOTA_BOOT_NONE;
89 /* A rollback saves sota_trial = 2 before it leaves the new image, so
90 * still running that image means power was lost in between. */
91 if (f->trial == SOTA_TRIAL_ROLLED_BACK) {
92 return f->running_part != 0 && f->prev_part != 0 && sota_on_new_image(f)
93 ? SOTA_BOOT_ROLLBACK_CUT
94 : SOTA_BOOT_REPORT_PENDING;
95 }
96 if (f->running_part == 0 || (f->new_part == 0 && f->prev_part == 0)) {
97 return SOTA_BOOT_INCOMPLETE;
98 }
99 bool on_new = sota_on_new_image(f);
100
101 /* Like the bootloader's probation, a trial ends at the first reset of
102 * the new image, whatever the cause: a crash, a watchdog, a power loss,
103 * a restart. Without that, an image that resets before it is confirmed
104 * gets a fresh confirm timeout on every boot, and never rolls back.
105 * Deep sleep counts too: where the bootloader resumes the image on a
106 * wake without validating it (CONFIG_BOOTLOADER_SKIP_VALIDATE_IN_DEEP_
107 * SLEEP), the trial would otherwise outlive any number of wakes. */
108 if (on_new) {
109 return f->trial == SOTA_TRIAL_BOOTED || f->deep_sleep_wake ? SOTA_BOOT_TRIAL_RESET
110 : SOTA_BOOT_TRIAL;
111 }
112
113 /* A deep-sleep wake can resume the old image while the boot slot already
114 * points at the new one, so the slot switch did happen. */
115 if (f->deep_sleep_wake && f->boot_part != 0 &&
116 (f->new_part != 0 ? f->boot_part == f->new_part
117 : f->boot_part != f->running_part)) {
118 return SOTA_BOOT_WAITING;
119 }
120 if (f->staged) return SOTA_BOOT_ABANDONED;
121 /* Any other reset that is not on the new image means it did not run,
122 * which is a rollback. */
123 return SOTA_BOOT_ROLLED_BACK;
124}
