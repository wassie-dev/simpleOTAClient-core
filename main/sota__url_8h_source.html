31static inline bool sota_url_has_scheme(const char *url, const char *scheme)
32{
33 for (; *scheme; ++url, ++scheme) {

55static inline sota_url_status_t sota_validate_server_url(const char *url)
56{
57 if (!url || !url[0]) return SOTA_URL_MALFORMED;
58 size_t len = strlen(url);
59 if (len > SOTA_SERVER_URL_MAX_LEN) return SOTA_URL_MALFORMED;
60 if (!sota_url_chars_ok(url)) return SOTA_URL_MALFORMED;
61 if (sota_url_has_scheme(url, "http://") && !sota_url_has_scheme(url, "https://"))
62 return SOTA_URL_NOT_HTTPS;
63 if (!sota_url_has_scheme(url, "https://")) return SOTA_URL_MALFORMED;
64 const char *slash = strchr(url + 8, '/');
65 return (slash ? (size_t)(slash - url) : len) == 8u ? SOTA_URL_MALFORMED : SOTA_URL_OK;
66}

81static inline sota_url_status_t sota_redirect_target(const char *base, const char *location,
82 char *out, size_t cap)
83{
84 if (!base || !location || !location[0] || !out || !sota_url_chars_ok(location)) {
85 return SOTA_URL_MALFORMED;
86 }
87 size_t origin = 0;
88 if (location[0] == '/' && location[1] != '/') {
89 if (!sota_url_has_scheme(base, "https://")) return SOTA_URL_NOT_HTTPS;
90 const char *slash = strchr(base + 8, '/');
91 origin = slash ? (size_t)(slash - base) : strlen(base);
92 } else if (sota_url_has_scheme(location, "http://")) {
93 return SOTA_URL_NOT_HTTPS;
94 } else if (!sota_url_has_scheme(location, "https://") || location[8] == '/' || !location[8]) {
95 return SOTA_URL_MALFORMED;
96 }
97 size_t len = strlen(location);
98 if (origin + len >= cap) return SOTA_URL_MALFORMED;
99 memcpy(out, base, origin);
100 memcpy(out + origin, location, len + 1);
101 return SOTA_URL_OK;
102}

117static inline bool sota_build_api_url(const char *base, const char *path, char *out,
118 size_t out_len)
119{
120 if (!base || !path || !out || out_len == 0) return false;
121 size_t n = strlen(base);
122 while (n > 8u && base[n - 1] == '/') --n;
123 size_t plen = strlen(path);
124 if (n >= out_len || plen > (out_len - 1u) - n) return false;
125 memcpy(out, base, n);
126 memcpy(out + n, path, plen + 1u);
127 return true;
128}
